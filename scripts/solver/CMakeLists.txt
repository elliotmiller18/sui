cmake_minimum_required(VERSION 3.15)
project(SuiSolver LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_OSX_DEPLOYMENT_TARGET 13.0)
get_filename_component(SOLVER "${CMAKE_CURRENT_LIST_DIR}/../../vendor/TexasSolver-console" ABSOLUTE)
file(GLOB SOURCES "${SOLVER}/src/*.cpp" "${SOLVER}/src/compairer/*.cpp" "${SOLVER}/src/nodes/*.cpp" "${SOLVER}/src/ranges/*.cpp" "${SOLVER}/src/tools/*.cpp" "${SOLVER}/src/trainable/*.cpp" "${SOLVER}/src/solver/*.cpp" "${SOLVER}/src/runtime/*.cpp")
list(FILTER SOURCES EXCLUDE REGEX "/CfrSolver\\.cpp$")
add_executable(console_solver ${SOURCES} "${SOLVER}/ext/fmt/src/format.cc")
target_include_directories(console_solver PRIVATE "${SOLVER}/include" "${SOLVER}/ext/fmt/include" "${OMP_INCLUDE}")
target_compile_options(console_solver PRIVATE -O3 -Xpreprocessor -fopenmp)
target_link_libraries(console_solver PRIVATE "${OMP_LIBRARY}")
